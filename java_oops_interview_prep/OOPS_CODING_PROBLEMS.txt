 HINTS (only if stuck)
====================================================================
 1-5  : Start with a class, one field per line, then the main.
 6    : Same method name, different parameters. Return type can vary.
 7    : static field -> one copy; access with ClassName.field.
 9    : Write @Override on each subclass method.
 10   : abstract methods have no body: void makeSound();
 12   : No setters. All fields assigned in the constructor.
 13   : hashCode = 31 * x + y style, or Objects.hash(x, y).
 14   : Composition: Car does `engine = new Engine();` inside.
        Aggregation: Department receives Employee[] in constructor.
 15   : private static final Logger INSTANCE with an enum or a
        holder class - keep it short.
 16   : Builder stores the same fields, build() does the copying.
 17   : Constructor takes a strategy; setter swaps it later.
 18   : publish() loops and calls notify() on every subscriber.
 19   : throw new InsufficientFundsException("message");
 20   : class Box<T> { T value; void set(T v){...} T get(){...} }
 21   : One class for data, one for printing, one for saving.
 22   : static blocks run first at class load, parents before
        children; instance blocks run before constructors.

SOLUTION 4
```java
class BankAccount {
    private double balance;

    void deposit(double amount) {
        if (amount <= 0) { System.out.println("Invalid deposit"); return; }
        balance += amount;
    }

    void withdraw(double amount) {
        if (amount <= 0 || amount > balance) {
            System.out.println("Withdrawal refused");
            return;
        }
        balance -= amount;
    }

    double getBalance() { return balance; }
}
public class Main {
    public static void main(String[] args) {
        BankAccount a = new BankAccount();
        a.deposit(100);
        a.withdraw(30);
        a.withdraw(1000);
        System.out.println("Balance: " + a.getBalance()); // 70.0
    }
}
```

class Shape {

class Point {
    int x, y;
    Point(int x, int y) { this.x = x; this.y = y; }

class Pizza {
    private final String size;
    private final String topping;
    private final boolean extraCheese;

    private Pizza(Builder b) {
        size = b.size;
        topping = b.topping;
        extraCheese = b.extraCheese;
    }

    static class Builder {
        private final String size;          // required
        private String topping = "none";    // optional
        private boolean extraCheese = false; // optional

        Builder(String size) { this.size = size; }

        Builder topping(String topping) { this.topping = topping; return this; }
        Builder extraCheese(boolean b) { extraCheese = b; return this; }

        Pizza build() { return new Pizza(this); }
    }

    public String toString() {
        return "Pizza(" + size + ", " + topping + ", cheese=" + extraCheese + ")";
    }
}
public class Main {
    public static void main(String[] args) {
        Pizza p = new Pizza.Builder("large")
                        .topping("mushroom")
                        .extraCheese(true)
                        .build();
        System.out.println(p);
    }

interface PaymentStrategy {
    void pay(int amount);
}
class CardPayment implements PaymentStrategy {
    public void pay(int amount) { System.out.println("Paid " + amount + " by card"); }
}
class UPIPayment implements PaymentStrategy {

    void setPaymentStrategy(PaymentStrategy strategy) {
        this.strategy = strategy;
    }

    void checkout(int amount) { strategy.pay(amount); }
}
public class Main {
    public static void main(String[] args) {
        ShoppingCart cart = new ShoppingCart();
        cart.setPaymentStrategy(new CardPayment());
        cart.checkout(100);
        cart.setPaymentStrategy(new UPIPayment());
        cart.checkout(50);
    }
}

```java
import java.util.ArrayList;
import java.util.List;

interface Subscriber {
    void notify(String news);
}
class EmailSubscriber implements Subscriber {
    public void notify(String news) { System.out.println("Email: " + news); }
}
class SMSSubscriber implements Subscriber {
    public void notify(String news) { System.out.println("SMS: " + news); }
}
class NewsPublisher {
    private List<Subscriber> subs = new ArrayList<>();

    void subscribe(Subscriber s) { subs.add(s); }

    void publish(String news) {
        for (Subscriber s : subs) {
            s.notify(news);
        }
    }
}
public class Main {
    public static void main(String[] args) {
        NewsPublisher publisher = new NewsPublisher();
        publisher.subscribe(new EmailSubscriber());
        publisher.subscribe(new SMSSubscriber());
        publisher.publish("Java 22 released");
    }
}
```

```java
class InsufficientFundsException extends Exception {
    InsufficientFundsException(String message) { super(message); }
}

class BankAccount {
    private double balance;

    BankAccount(double balance) { this.balance = balance; }

    void withdraw(double amount) throws InsufficientFundsException {
        if (amount > balance) {
            throw new InsufficientFundsException("Not enough money");
        }
        balance -= amount;
    }
}
public class Main {
    public static void main(String[] args) {
        BankAccount a = new BankAccount(50);
        try {
            a.withdraw(100);
        } catch (InsufficientFundsException e) {
            System.out.println("Caught: " + e.getMessage());
        }
    }
}
```

SOLUTION 20
```java
import java.util.Comparator;

class Box<T> {
    private T value;

    void set(T value) { this.value = value; }
    T get() { return value; }
}

public class Main {
    static <T> T max2(T a, T b, Comparator<T> c) {
        return c.compare(a, b) >= 0 ? a : b;
    }

    public static void main(String[] args) {
        Box<String> box = new Box<>();
        box.set("hello");
        System.out.println(box.get());

        String bigger = max2("apple", "zebra", (x, y) -> x.compareTo(y));
        System.out.println(bigger); // zebra
    }
}
```
