10. TIMER (setInterval)
------------------------------------------------------------
QUESTION: "Show a running timer in seconds."
WHAT TO BUILD: seconds counter + start/pause.
CONCEPTS TESTED: useEffect, cleanup, setInterval.
EXPECTED UI: 0, 1, 2... every second.
HINTS: interval in useEffect with [] deps; clearInterval in cleanup;
      use the functional update to avoid stale state.
REFERENCE SOLUTION:
  function App() {
    const [seconds, setSeconds] = useState(0);
    const [running, setRunning] = useState(false);
    useEffect(() => {
      if (!running) return;
      const id = setInterval(() => setSeconds((s) => s + 1), 1000);
      return () => clearInterval(id);
    }, [running]);
    return (
      <div>
        <h1>{seconds}s</h1>
        <button onClick={() => setRunning(!running)}>
          {running ? "Pause" : "Start"}
        </button>
      </div>
    );
  }

      <div>

          <button>Add</button>
        </form>
        <ul>
          {todos.map((t) => (
            <li key={t.id}>
              {t.text}
              <button onClick={() => setTodos(todos.filter((x) => x.id !== t.id))}>
                Delete
              </button>
            </li>
          ))}
        </ul>
      </div>
    );
  }

12. TODO — TOGGLE COMPLETE
------------------------------------------------------------
QUESTION: "Add a strikethrough when a todo is completed."
WHAT TO BUILD: same as 11 + click-to-toggle.
CONCEPTS TESTED: map + spread update.
EXPECTED UI: clicking an item strikes it through; clicking again unstrikes.
HINTS: map and replace the matched item; style={{textDecoration}}.
REFERENCE SOLUTION:
  function App() {
    const [todos, setTodos] = useState([]);
    const [text, setText] = useState("");
    const add = (e) => {
      e.preventDefault();
      if (!text.trim()) return;
      setTodos([...todos, { id: Date.now(), text, done: false }]);
      setText("");
    };
    const toggle = (id) =>
      setTodos(todos.map((t) =>
        t.id === id ? { ...t, done: !t.done } : t));
    return (
      <div>
        <form onSubmit={add}>
          <input value={text} onChange={(e) => setText(e.target.value)} />
          <button>Add</button>
        </form>
        <ul>
          {todos.map((t) => (
            <li key={t.id}
                onClick={() => toggle(t.id)}
                style={{ textDecoration: t.done ? "line-through" : "none" }}>
              {t.text}
            </li>
          ))}
        </ul>
      </div>
    );
  }

21. SHOPPING CART
------------------------------------------------------------
QUESTION: "A product list with an add-to-cart cart and total."
WHAT TO BUILD: products + cart list + total price.
CONCEPTS TESTED: state objects, array updates, reduce.
EXPECTED UI: Add -> item in cart; total updates; Remove works.
HINTS: cart as array of {id, qty}; total = reduce.
REFERENCE SOLUTION:
  const PRODUCTS = [
    { id: 1, name: "Apple", price: 30 },
    { id: 2, name: "Milk", price: 55 },
    { id: 3, name: "Bread", price: 40 },
  ];

  function App() {
    const [cart, setCart] = useState([]);   // [{id, qty}]
    const add = (product) => {
      const found = cart.find((c) => c.id === product.id);
      if (found) {
        setCart(cart.map((c) =>
          c.id === product.id ? { ...c, qty: c.qty + 1 } : c));
      } else {
        setCart([...cart, { id: product.id, qty: 1 }]);
      }
    };
    const remove = (id) =>
      setCart(cart.filter((c) => c.id !== id));

    const total = cart.reduce((sum, c) => {
      const p = PRODUCTS.find((p) => p.id === c.id);
      return sum + p.price * c.qty;
    }, 0);

    return (
      <div>
        {PRODUCTS.map((p) => (
          <div key={p.id}>
            {p.name} — Rs {p.price}
            <button onClick={() => add(p)}>Add</button>
          </div>
        ))}
        <h2>Cart (total Rs {total})</h2>
        {cart.map((c) => (
          <div key={c.id}>
            {PRODUCTS.find((p) => p.id === c.id).name} x{c.qty}
            <button onClick={() => remove(c.id)}>Remove</button>
          </div>
        ))}
      </div>
    );
  }

26. SEARCH + PAGINATION COMBINED
------------------------------------------------------------
QUESTION: "Filter a list AND paginate the filtered result."
WHAT TO BUILD: search box + prev/next over the filtered items.
CONCEPTS TESTED: filter, slice, derived values.
EXPECTED UI: typing resets to page 1 and pages shrink.
HINTS: derive filtered first, then slice it. Reset page to 1 in onChange.
REFERENCE SOLUTION:
  function App() {
    const all = Array.from({ length: 60 }, (_, i) => `User ${i + 1}`);
    const [q, setQ] = useState("");
    const [page, setPage] = useState(1);
    const PAGE_SIZE = 5;

    const filtered = all.filter((u) =>
      u.toLowerCase().includes(q.toLowerCase()));
    const totalPages = Math.ceil(filtered.length / PAGE_SIZE) || 1;
    const items = filtered.slice((page - 1) * PAGE_SIZE, page * PAGE_SIZE);

    const changeQ = (e) => { setQ(e.target.value); setPage(1); };

    return (
      <div>
        <input value={q} onChange={changeQ} />
        <ul>{items.map((i) => <li key={i}>{i}</li>)}</ul>
        <button disabled={page === 1} onClick={() => setPage(page - 1)}>Prev</button>
        <span>{page} / {totalPages}</span>
        <button disabled={page === totalPages} onClick={() => setPage(page + 1)}>Next</button>
      </div>
    );
  }

30. CUSTOM HOOK — useFetch
------------------------------------------------------------
QUESTION: "Write a useFetch hook and use it to load data."
WHAT TO BUILD: hook + one consumer.
CONCEPTS TESTED: custom hooks, useState/useEffect inside a hook.
EXPECTED UI: loading/error/data, same as 17.
HINTS: hook returns { data, loading, error }; state lives in the hook.
REFERENCE SOLUTION:
  function useFetch(url) {
    const [data, setData] = useState(null);
    const [loading, setLoading] = useState(true);
    const [error, setError] = useState("");

    useEffect(() => {
      fetch(url)
        .then((r) => r.json())
        .then(setData)
        .catch((e) => setError(e.message))
        .finally(() => setLoading(false));
    }, [url]);

    return { data, loading, error };
  }

  function App() {
    const { data, loading, error } = useFetch(
      "https://jsonplaceholder.typicode.com/users");
    if (loading) return <p>Loading...</p>;
    if (error) return <p>{error}</p>;
    return <ul>{data.map((u) => <li key={u.id}>{u.name}</li>)}</ul>;
  }

31. COMPLEX FORM STATE (useReducer)
------------------------------------------------------------
QUESTION: "Manage a registration form with useReducer."
WHAT TO BUILD: 4 fields + errors via a reducer.
CONCEPTS TESTED: useReducer, actions.
EXPECTED UI: same as a normal form, but state updates go through dispatch.
HINTS: reducer actions: SET_FIELD, SET_ERRORS, RESET.
REFERENCE SOLUTION:
  const initial = { values: {}, errors: {} };

  function reducer(state, action) {
    switch (action.type) {
      case "SET_FIELD":
        return { ...state, values: { ...state.values, [action.name]: action.value } };
      case "SET_ERRORS":
        return { ...state, errors: action.errors };
      case "RESET":
        return initial;
      default:
        return state;
    }
  }

  function App() {
    const [state, dispatch] = useReducer(reducer, initial);

    const submit = (e) => {
      e.preventDefault();
      const errs = {};
      if (!state.values.email) errs.email = "Email required";
      if (!state.values.password) errs.password = "Password required";
      dispatch({ type: "SET_ERRORS", errors: errs });
    };

    return (
      <form onSubmit={submit}>
        <input name="email" placeholder="Email"
          onChange={(e) => dispatch({ type: "SET_FIELD", name: "email", value: e.target.value })} />
        {state.errors.email && <p className="error">{state.errors.email}</p>}
        <input name="password" type="password" placeholder="Password"
          onChange={(e) => dispatch({ type: "SET_FIELD", name: "password", value: e.target.value })} />
        {state.errors.password && <p className="error">{state.errors.password}</p>}
        <button>Submit</button>
      </form>
    );
  }

35. DEBOUNCED SEARCH WITH API
------------------------------------------------------------
QUESTION: "Debounce an API search instead of calling on every keystroke."
WHAT TO BUILD: input -> debounced query -> fetch results.
CONCEPTS TESTED: debounce + fetch + loading.
EXPECTED UI: results arrive 500ms after typing stops.
HINTS: two effects — one debounces, one fetches on the debounced value.
REFERENCE SOLUTION:
  function App() {
    const [query, setQuery] = useState("");
    const [debounced, setDebounced] = useState("");
    const [results, setResults] = useState([]);
    const [loading, setLoading] = useState(false);

    useEffect(() => {
      const t = setTimeout(() => setDebounced(query), 500);
      return () => clearTimeout(t);
    }, [query]);

    useEffect(() => {
      if (!debounced) return;
      setLoading(true);
      fetch(`https://jsonplaceholder.typicode.com/posts?userId=${debounced.length}`)
        .then((r) => r.json())
        .then(setResults)
        .finally(() => setLoading(false));
    }, [debounced]);

    return (
      <div>
        <input value={query} onChange={(e) => setQuery(e.target.value)}
          placeholder="Search (debounced)" />
        {loading && <p>Searching...</p>}
        <ul>{results.slice(0, 5).map((r) => <li key={r.id}>{r.title}</li>)}</ul>
      </div>
    );
  }

40. BUILD A PAGE FROM SCRATCH
------------------------------------------------------------
QUESTION: "Build a products page: fetch products, search, add to cart,
           show cart count and total."
WHAT TO BUILD: full small app combining many patterns.
CONCEPTS TESTED: fetch, loading/error, search, cart state, reduce.
EXPECTED UI: loading -> searchable product list -> cart summary.
HINTS: reuse patterns 17 + 34 + 21. Build top-down: App owns
      products and cart; ProductList and Cart are presentational.
REFERENCE SOLUTION:
  function App() {
    const [products, setProducts] = useState([]);
    const [loading, setLoading] = useState(true);
    const [q, setQ] = useState("");
    const [cart, setCart] = useState([]);

    useEffect(() => {
      fetch("https://fakestoreapi.com/products?limit=10")
        .then((r) => r.json())
        .then(setProducts)
        .finally(() => setLoading(false));
    }, []);

    const filtered = products.filter((p) =>
      p.title.toLowerCase().includes(q.toLowerCase()));

    const addToCart = (p) => {
      const found = cart.find((c) => c.id === p.id);
      setCart(found
        ? cart.map((c) => c.id === p.id ? { ...c, qty: c.qty + 1 } : c)
        : [...cart, { id: p.id, qty: 1 }]);
    };

    const total = cart.reduce((s, c) => {
      const p = products.find((p) => p.id === c.id);
      return s + (p ? p.price * c.qty : 0);
    }, 0);

    if (loading) return <p>Loading...</p>;

    return (
      <div>
        <h1>Cart: {cart.reduce((s, c) => s + c.qty, 0)} items — Rs {total.toFixed(2)}</h1>
        <input value={q} onChange={(e) => setQ(e.target.value)} placeholder="Search" />
        {filtered.map((p) => (
          <div key={p.id}>
            {p.title}
            <button onClick={() => addToCart(p)}>Add</button>
          </div>
        ))}
      </div>
    );
  }
